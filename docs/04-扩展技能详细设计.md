### 1.1 Claim 相关模型（Pydantic）

```python
# src/jobpilot/models/claim.py

from enum import Enum
from typing import Optional
from pydantic import BaseModel, Field
from datetime import datetime

class ClaimType(str, Enum):
    """Claim 五分类"""
    OWNERSHIP = "Ownership"      # 个人负责/主导
    METRIC = "Metric"            # 可量化指标
    TECHNICAL = "Technical"      # 技术能力/工具
    ARCHITECTURE = "Architecture" # 系统设计/架构
    RESULT = "Result"            # 成果/影响

class ClaimStrength(str, Enum):
    """证据强度"""
    STRONG = "strong"    # 强证据：明确、具体、可验证
    WEAK = "weak"        # 弱证据：模糊、笼统、缺乏细节
    MISSING = "missing"  # 缺失：无证据

class TextSpan(BaseModel):
    """文本定位"""
    start: int = Field(..., description="起始字符位置")
    end: int = Field(..., description="结束字符位置")
    text: str = Field(..., description="原文内容")

class GapType(str, Enum):
    """Gap 分类"""
    HARD_BLOCK = "hard_block"  # 硬性门槛不满足（如学历、必须技能）
    SOFT_GAP = "soft_gap"      # 软性差距（如加分项、经验不足）

class Claim(BaseModel):
    """可验证声明"""
    id: str = Field(..., description="Claim ID，格式：claim_{job_id}_{序号}")
    claim_type: ClaimType
    claim_text: str = Field(..., description="声明内容")
    resume_span: TextSpan = Field(..., description="简历原文定位")
    jd_evidence: Optional[str] = Field(None, description="JD 中对应的证据")
    jd_span: Optional[TextSpan] = Field(None, description="JD 原文定位")
    strength: ClaimStrength

class Gap(BaseModel):
    """JD 要求与简历的差距"""
    id: str = Field(..., description="Gap ID，格式：gap_{job_id}_{序号}")
    gap_type: GapType
    jd_requirement: str = Field(..., description="JD 中的要求原文")
    jd_span: TextSpan = Field(..., description="JD 原文定位")
    related_claims: list[str] = Field(default_factory=list, description="关联的 Claim ID 列表")
    mitigation: Optional[str] = Field(None, description="弥补建议（soft_gap 时必填）")

class EvidenceMatrix(BaseModel):
    """证据矩阵：Claim 与 JD 要求的映射关系"""
    claim_id: str
    jd_requirement_id: str
    match_status: str = Field(..., description="匹配状态：matched/partial/missing")

class ClaimExtractionResult(BaseModel):
    """Claim 提取结果"""
    id: str = Field(..., description="提取结果 ID")
    job_id: str
    resume_version: str
    claims: list[Claim]
    gaps: list[Gap]
    evidence_matrix: list[EvidenceMatrix]
    extraction_tokens: int = Field(0, description="提取消耗的 token 数")
    extraction_cost: float = Field(0.0, description="提取消耗的费用（元）")
    created_at: datetime = Field(default_factory=datetime.now)
```

```python
# src/jobpilot/tools/claim_extractor_tool.py

from typing import Optional
from crewai.tools import BaseTool
from pydantic import BaseModel

class ClaimExtractorInput(BaseModel):
    """Claim 提取输入"""
    resume_text: str
    jd_text: str
    parsed_jd: dict  # 来自 Parser 的结构化 JD
    resume_version: str
    job_id: str

class ClaimExtractorTool(BaseTool):
    """Claim 提取工具"""
    name: str = "claim_extractor"
    description: str = "从简历中提取可验证声明，与 JD 要求进行匹配，识别差距"
    args_schema: type[BaseModel] = ClaimExtractorInput

    def _run(
        self,
        resume_text: str,
        jd_text: str,
        parsed_jd: dict,
        resume_version: str,
        job_id: str
    ) -> dict:
        """
        执行 Claim 提取

        Returns:
            dict: {
                "claims": Claim[],
                "gaps": Gap[],
                "evidence_matrix": EvidenceMatrix[],
                "extraction_tokens": int,
                "extraction_cost": float
            }
        """
        # 1. 检查缓存
        cache_key = self._compute_cache_key(jd_text, resume_version)
        cached = self._check_cache(cache_key)
        if cached:
            return cached

        # 2. 构建 prompt
        prompt = self._build_prompt(resume_text, jd_text, parsed_jd)

        # 3. 调用 LLM（通过网关）
        result = self._call_llm(prompt)

        # 4. 解析并验证输出
        parsed_result = self._parse_and_validate(result)

        # 5. 写入缓存
        self._save_to_cache(cache_key, parsed_result)

        # 6. 写入数据库
        self._save_to_db(parsed_result, job_id, resume_version)

        return parsed_result
```

## 三、算法详细设计

### 3.1 Claim 提取算法

#### 3.1.1 提取流程

```
输入：resume_text, jd_text, parsed_jd
输出：ClaimExtractionResult

1. 文本预处理
   - 清理简历文本（去除格式标记、特殊字符）
   - 分段（按经历/项目/技能分类）

2. Claim 识别（LLM）
   - 输入：简历分段文本
   - 输出：Claim[]（claim_type, claim_text, resume_span）
   - 约束：每个 Claim 必须有明确的 resume_span 定位

3. JD 要求提取
   - 从 parsed_jd 中提取所有要求项
   - 为每个要求项生成唯一 ID（jd_req_{序号}）

4. Claim-JD 映射（LLM）
   - 输入：Claim[] + JD 要求项[]
   - 输出：EvidenceMatrix[]（claim_id, jd_req_id, match_status）
   - 匹配策略：
     * matched：Claim 直接满足 JD 要求
     * partial：Claim 部分满足 JD 要求
     * missing：Claim 与 JD 要求无关

5. Gap 检测
   - 遍历 JD 要求项
   - 如果无 Claim 映射到该要求项，或所有映射都是 partial/missing，则为 Gap
   - Gap 分类规则：
     * hard_block：要求中包含"必须"、"required"、"mandatory"等关键词
     * soft_gap：其他情况

6. 输出组装
   - 为每个 Claim 和 Gap 生成唯一 ID
   - 计算 extraction_tokens 和 extraction_cost
   - 返回 ClaimExtractionResult
```

#### 3.1.2 缓存策略

```python
def _compute_cache_key(self, jd_text: str, resume_version: str) -> str:
    """计算缓存键"""
    import hashlib
    content = f"{jd_text}|{resume_version}"
    return hashlib.sha256(content.encode()).hexdigest()
```

**合并调用策略**：Claim 提取与评分合并为一次 LLM 调用，减少延迟和成本。

### 3.2 Gap Advisor 算法

#### 3.2.1 建议生成流程

```
输入：resume_text, job_posting_summary, claims_json, gaps_json
输出：ResumePatch

1. Gap 分析
   - 将 gaps_json 按 gap_type 分组
   - 对每个 gap，关联 related_claims

2. Bullet 改写（LLM）
   - 输入：resume_text + gaps_json + claims_json
   - 输出：BulletRewrite[]
   - 改写策略：
     * 对 soft_gap，改写相关经历的表述，突出与 JD 要求的匹配点
     * 保留原有 Claim 的真实性，不夸大

3. 缺失证据建议（LLM）
   - 输入：gaps_json（仅 soft_gap）
   - 输出：MissingEvidence[]
   - 建议策略：
     * 对 high priority 的 gap，建议具体的项目/活动
     * 对 medium/low priority 的 gap，建议学习方向

4. HR 开场白生成（LLM）
   - 输入：job_posting_summary + claims_json（选择 strength=strong 的 Claim）
   - 输出：HROpener
   - 生成策略：
     * 选择 2-3 个与 JD 最相关的 strong Claim
     * 生成一段话，突出个人优势与职位匹配度

5. ResumePatch 组装
   - 将 BulletRewrite[]、MissingEvidence[]、HROpener 组装为 ResumePatch
   - 生成 patch_markdown（可直接合并的增量内容）
   - 计算 generation_tokens 和 generation_cost
```

#### 3.2.2 触发条件

```python
def should_generate_patch(self, match_score: MatchScore) -> bool:
    """判断是否生成 ResumePatch"""
    return match_score.overall >= 70
```

### 3.3 Interview Prep 算法

#### 3.3.1 生成流程

```
输入：job_posting_summary, company_research, claims_json, gaps_json, resume_text
输出：InterviewBrief

1. 问题预测（LLM）
   - 输入：job_posting_summary + claims_json
   - 输出：PredictedQuestion[]
   - 预测策略：
     * jd_direct：JD 明确要求的问题（如"有 XX 经验吗"）
     * jd_inferred：从 JD 推断的问题（如 JD 要求 Python，可能问 Python 细节）
     * company_culture：公司文化相关问题（从 company_research 提取）

2. 追问协议生成（LLM）
   - 输入：PredictedQuestion[] + claims_json
   - 输出：FollowUp[]
   - 追问策略：
     * 对 Metric 类 Claim，追问具体数字和计算方法
     * 对 Ownership 类 Claim，追问个人贡献和决策过程
     * 对 Technical 类 Claim，追问技术细节和踩坑经历

3. 弱项练习生成（LLM）
   - 输入：gaps_json（strength=weak/missing 的 Claim）
   - 输出：WeakAreaDrill[]
   - 练习策略：
     * 对 weak Claim，生成巩固练习题
     * 对 missing Claim，生成学习方向和练习题

4. 面试速览组装
   - 整合 company_research、tech_stack、PredictedQuestion[]、FollowUp[]、WeakAreaDrill[]
   - 生成 brief_markdown（一页纸）
   - 计算 generation_tokens 和 generation_cost
```

#### 3.3.2 公司调研集成

```python
def _get_company_research(self, company: str) -> str:
    """获取公司调研报告"""
    # 复用 Researcher Agent 的输出
    if self.researcher_cache.has(company):
        return self.researcher_cache.get(company)

    # 调用 Researcher Agent
    research = self.researcher.research(company)
    self.researcher_cache.set(company, research)
    return research
```

```python
from typing import Optional, List
from .base_dao import BaseDAO

class ResumePatchDAO(BaseDAO):
    """Resume Patch DAO"""

    def save(self, patch: ResumePatch) -> str:
        """保存 Resume Patch"""
        sql = """
        INSERT INTO resume_patches
            (id, job_id, bullet_rewrites_json, missing_evidence_json,
             hr_opener_text, patch_markdown, generation_tokens, generation_cost, created_at)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        """
        self.execute(sql, (
            patch.id, patch.job_id,
            patch.bullet_rewrites.json(), patch.missing_evidence.json(),
            patch.hr_opener.opener_text if patch.hr_opener else None,
            patch.patch_markdown,
            patch.generation_tokens, patch.generation_cost,
            patch.created_at
        ))
        return patch.id

    def get_by_job_id(self, job_id: str) -> Optional[ResumePatch]:
        """根据 job_id 获取 Resume Patch"""
        sql = "SELECT * FROM resume_patches WHERE job_id = ? ORDER BY created_at DESC LIMIT 1"
        row = self.fetch_one(sql, (job_id,))
        if row:
            return self._row_to_model(row)
        return None

    def get_top_n(self, n: int = 10) -> List[ResumePatch]:
        """获取 Top N Resume Patch（按 job 匹配分数排序）"""
        sql = """
        SELECT rp.* FROM resume_patches rp
        JOIN match_scores ms ON rp.job_id = ms.job_id
        ORDER BY ms.overall DESC
        LIMIT ?
        """
        rows = self.fetch_all(sql, (n,))
        return [self._row_to_model(row) for row in rows]
```

```python
            brief.id, brief.job_id, brief.company_overview,
            brief.tech_stack.json(),
            brief.predicted_questions.json(), brief.followup_protocol.json(),
            brief.weak_area_drills.json(), brief.evidence_summary.json(),
            brief.brief_markdown,
            brief.generation_tokens, brief.generation_cost,
            brief.created_at
        ))
        return brief.id

    def get_by_job_id(self, job_id: str) -> Optional[InterviewBrief]:
        """根据 job_id 获取 Interview Brief"""
        sql = "SELECT * FROM interview_briefs WHERE job_id = ? ORDER BY created_at DESC LIMIT 1"
        row = self.fetch_one(sql, (job_id,))
        if row:
            return self._row_to_model(row)
        return None

    def get_top_n(self, n: int = 5) -> List[InterviewBrief]:
        """获取 Top N Interview Brief（按 job 匹配分数排序）"""
        sql = """
        SELECT ib.* FROM interview_briefs ib
        JOIN match_scores ms ON ib.job_id = ms.job_id
        ORDER BY ms.overall DESC
        LIMIT ?
        """
        rows = self.fetch_all(sql, (n,))
        return [self._row_to_model(row) for row in rows]
```

```

   - jd_inferred：从 JD 推断的问题（如 JD 要求 Python，可能问"Python 的 GIL 是什么"）
   - company_culture：公司文化相关问题（从 company_research 提取，如"为什么选择我们公司"）

   - 对 Metric 类 Claim，追问具体数字和计算方法（如"50% 是怎么算出来的"）

   - 对 strength=weak 的 Claim，生成巩固练习题
   - 对 strength=missing 的 Claim，生成学习方向和练习题
   - 每个练习题给出期望的回答证据和常见错误

   - 预测问题（按重要性排序，Top 5）
   - 证据摘要（按 Claim 类型分组，列出你的优势）
   - 弱项提示（需要准备的 Gap）

- 追问要能检测 Claim 的真实性，不能是简单的"还有吗"
- 弱项练习要给出具体的学习方向，不能是"多学习"
- brief_markdown 要简洁，控制在一页纸内（约 500 字）
- 输出必须是合法的 JSON 格式
```
