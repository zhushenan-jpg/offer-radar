## 一、测试策略总览

### 1.1 测试层次

| 层次 | 目标 | 工具 | 覆盖率目标 |
|------|------|------|------------|
| 单元测试 | 验证单个函数/方法的正确性 | pytest | ≥ 90% |
| 集成测试 | 验证模块间协作 | pytest + mock | ≥ 80% |
| 端到端测试 | 验证完整业务流程 | pytest + 真实 LLM（可选） | 核心流程 100% |
| 人工评测 | 验证输出质量 | 人工标注 + 评测脚本 | 50 条样本 |

### 1.2 测试原则

1. **LLM 调用一律 mock**：单元测试和集成测试中不调用真实 LLM API
2. **测试数据可复现**：所有测试数据固定，不依赖外部状态
3. **测试独立**：每个测试用例独立运行，不依赖其他测试的执行结果
4. **测试可读**：测试用例名称清晰描述测试场景和预期结果

import pytest
from pydantic import ValidationError
from jobpilot.models.claim import (
    ClaimType, ClaimStrength, GapType,
    TextSpan, Claim, Gap, EvidenceMatrix, ClaimExtractionResult
)

class TestClaimType:
    """Claim 类型枚举测试"""

    def test_claim_type_values(self):
        """验证 5 种 Claim 类型值正确"""
        assert ClaimType.OWNERSHIP == "Ownership"
        assert ClaimType.METRIC == "Metric"
        assert ClaimType.TECHNICAL == "Technical"
        assert ClaimType.ARCHITECTURE == "Architecture"
        assert ClaimType.RESULT == "Result"

    def test_claim_type_from_string(self):
        """验证从字符串创建 Claim 类型"""
        assert ClaimType("Ownership") == ClaimType.OWNERSHIP
        assert ClaimType("Metric") == ClaimType.METRIC

    def test_claim_type_invalid(self):
        """验证无效 Claim 类型抛出异常"""
        with pytest.raises(ValueError):
            ClaimType("InvalidType")

class TestTextSpan:
    """文本定位测试"""

    def test_valid_span(self):
        """验证有效文本定位"""
        span = TextSpan(start=0, end=10, text="hello world")
        assert span.start == 0
        assert span.end == 10
        assert span.text == "hello world"

    def test_span_start_negative(self):
        """验证起始位置为负数"""
        with pytest.raises(ValidationError):
            TextSpan(start=-1, end=10, text="test")

    def test_span_end_before_start(self):
        """验证结束位置小于起始位置"""
        with pytest.raises(ValidationError):
            TextSpan(start=10, end=5, text="test")

class TestClaim:
    """Claim 模型测试"""

    def test_valid_claim(self):
        """验证有效 Claim"""
        claim = Claim(
            id="claim_001",
            claim_type=ClaimType.METRIC,
            claim_text="将 API 响应时间从 2s 优化到 200ms",
            resume_span=TextSpan(start=0, end=30, text="优化了 API 响应时间"),
            jd_evidence="要求有性能优化经验",
            jd_span=TextSpan(start=100, end=120, text="具备性能优化经验"),
            strength=ClaimStrength.STRONG
        )
        assert claim.id == "claim_001"
        assert claim.claim_type == ClaimType.METRIC
        assert claim.strength == ClaimStrength.STRONG

    def test_claim_without_jd_evidence(self):
        """验证无 JD 证据的 Claim"""
        claim = Claim(
            id="claim_002",
            claim_type=ClaimType.TECHNICAL,
            claim_text="使用 Python 开发后端",
            resume_span=TextSpan(start=0, end=20, text="使用 Python 开发"),
            strength=ClaimStrength.WEAK
        )
        assert claim.jd_evidence is None
        assert claim.jd_span is None

    def test_claim_missing_required_field(self):
        """验证缺少必填字段"""
        with pytest.raises(ValidationError):
            Claim(
                claim_type=ClaimType.METRIC,
                claim_text="test"
                # 缺少 id, resume_span, strength
            )

class TestGap:
    """Gap 模型测试"""

    def test_hard_block_gap(self):

import pytest
import json
from jobpilot.eval.claim_evaluator import ClaimEvaluator

class TestClaimAccuracy:
    """Claim 提取准确率评测"""

    @pytest.fixture
    def evaluator(self):
        return ClaimEvaluator()

    @pytest.fixture
    def annotated_samples(self):
        """加载人工标注样本"""
        with open("tests/fixtures/annotated_claims.json", "r") as f:
            return json.load(f)

    def test_claim_type_accuracy(self, evaluator, annotated_samples):
        """测试 Claim 分类准确率"""
        correct = 0
        total = len(annotated_samples)

        for sample in annotated_samples:
            predicted = evaluator.predict_claim_type(
                sample["resume_text"],
                sample["claim_text"]
            )
            if predicted == sample["expected_type"]:
                correct += 1

        accuracy = correct / total
        assert accuracy >= 0.85, f"Claim 分类准确率 {accuracy:.2%} 低于 85% 目标"

    def test_gap_detection_recall(self, evaluator, annotated_samples):
        """测试 Gap 检测召回率"""
        true_gaps = [s for s in annotated_samples if s.get("is_gap")]
        detected_gaps = evaluator.detect_gaps(
            [s["resume_text"] for s in annotated_samples],
            [s["jd_text"] for s in annotated_samples]
        )

        recall = len(detected_gaps) / len(true_gaps) if true_gaps else 1.0
        assert recall >= 0.80, f"Gap 检测召回率 {recall:.2%} 低于 80% 目标"

        return QuestionEvaluator()

    @pytest.fixture
    def annotated_samples(self):

        with open("tests/fixtures/annotated_questions.json", "r") as f:

    def test_question_coverage(self, evaluator, annotated_samples):

                requirements, predicted
            )

### 8.1 评测频率决策

**决策**：人工评测在里程碑节点跑，prompt 微调用自动化回归测试覆盖。

**理由**：
- 人工评测成本高（需要标注 50 条样本），不适合每次 prompt 改动都跑
- 自动化回归测试（基于 mock 数据）可以快速验证 prompt 改动不会导致输出格式异常
- 里程碑节点（M5/M6/M7）跑人工评测，确保质量达标

**执行策略**：
- **日常开发**：跑单元测试 + 集成测试（mock 数据，< 1 分钟）
- **Prompt 改动**：跑自动化回归测试（mock 数据 + 格式校验，< 5 分钟）
- **里程碑节点**：跑人工评测（50 条样本，需要 1-2 小时标注时间）
