### 1.2 Prompt 模板维护

#### 1.2.1 修改流程

```
1. 修改 prompts/ 目录下的 Prompt 模板文件
2. 运行自动化回归测试：pytest tests/eval/test_regression.py
3. 如果回归测试通过，提交代码
4. 如果回归测试失败，检查失败原因：
   - 输出格式异常 → 调整 Prompt 的格式约束
   - 质量下降 → 回滚 Prompt 或调整 few-shot 示例
5. 里程碑节点运行人工评测，确认质量达标
```

```python
# scripts/migrate_v1_to_v2.py

"""数据库迁移脚本：新增扩展技能表"""

import sqlite3

def migrate(db_path: str):
    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()

    # 新增 claim_extractions 表
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS claim_extractions (
        id TEXT PRIMARY KEY,
        job_id TEXT NOT NULL,
        resume_version TEXT NOT NULL,
        claims_json TEXT NOT NULL,
        gaps_json TEXT NOT NULL,
        evidence_matrix_json TEXT,
        extraction_tokens INTEGER,
        extraction_cost REAL,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY (job_id) REFERENCES job_postings(id)
    )
    """)

    # 新增 resume_patches 表
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS resume_patches (
        id TEXT PRIMARY KEY,
        job_id TEXT NOT NULL,
        bullet_rewrites_json TEXT,
        missing_evidence_json TEXT,
        hr_opener_text TEXT,
        patch_markdown TEXT,
        generation_tokens INTEGER,
        generation_cost REAL,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY (job_id) REFERENCES job_postings(id)
    )
    """)

    # 新增 interview_briefs 表
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS interview_briefs (
        id TEXT PRIMARY KEY,
        job_id TEXT NOT NULL,
        company_overview TEXT,
        tech_stack_json TEXT,
        predicted_questions_json TEXT,
        followup_protocol_json TEXT,
        weak_area_drills_json TEXT,
        evidence_summary_json TEXT,
        brief_markdown TEXT,
        generation_tokens INTEGER,
        generation_cost REAL,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY (job_id) REFERENCES job_postings(id)
    )
    """)

    # 扩展 match_scores 表
    cursor.execute("ALTER TABLE match_scores ADD COLUMN claims_json TEXT")
    cursor.execute("ALTER TABLE match_scores ADD COLUMN gaps_json TEXT")
    cursor.execute("ALTER TABLE match_scores ADD COLUMN evidence_matrix_json TEXT")

    conn.commit()
    conn.close()

if __name__ == "__main__":
    import sys
    migrate(sys.argv[1])
```

```python
# src/jobpilot/storage/cache_manager.py

class CacheManager:
    """缓存管理器"""

    def __init__(self, db_path: str):
        self.db_path = db_path

    def cleanup_expired(self, days: int = 30):
        """清理过期缓存"""
        import sqlite3
        from datetime import datetime, timedelta

        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        cutoff = datetime.now() - timedelta(days=days)

        # 清理 claim_extractions 缓存
        cursor.execute("""
        DELETE FROM claim_extractions
        WHERE created_at < ?
        """, (cutoff,))

        # 清理 resume_patches 缓存
        cursor.execute("""
        DELETE FROM resume_patches
        WHERE created_at < ?
        """, (cutoff,))

        # 清理 interview_briefs 缓存
        cursor.execute("""
        DELETE FROM interview_briefs
        WHERE created_at < ?
        """, (cutoff,))

        conn.commit()
        conn.close()

    def get_cache_stats(self) -> dict:
        """获取缓存统计"""
        import sqlite3

        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        stats = {}

        cursor.execute("SELECT COUNT(*) FROM claim_extractions")
        stats["claim_extractions"] = cursor.fetchone()[0]

        cursor.execute("SELECT COUNT(*) FROM resume_patches")
        stats["resume_patches"] = cursor.fetchone()[0]

        cursor.execute("SELECT COUNT(*) FROM interview_briefs")
        stats["interview_briefs"] = cursor.fetchone()[0]

        conn.close()
        return stats
```

#### 2.2.1 指标采集器

```python
# src/jobpilot/monitoring/metrics_collector.py

from dataclasses import dataclass
from datetime import datetime
from typing import Optional
import sqlite3

@dataclass
class MetricPoint:
    """指标数据点"""
    name: str
    value: float
    labels: dict
    timestamp: datetime

class MetricsCollector:
    """指标采集器"""

    def __init__(self, db_path: str):
        self.db_path = db_path
        self._init_metrics_table()

    def _init_metrics_table(self):
        """初始化指标表"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("""
        CREATE TABLE IF NOT EXISTS metrics (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            value REAL NOT NULL,
            labels_json TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
        """)
        conn.commit()
        conn.close()

    def record_latency(self, operation: str, latency_ms: float, job_id: str):
        """记录延迟指标"""
        self._insert_metric(
            name=f"{operation}_latency_ms",
            value=latency_ms,
            labels={"job_id": job_id}
        )

    def record_cost(self, operation: str, cost_yuan: float, job_id: str):
        """记录成本指标"""
        self._insert_metric(
            name=f"{operation}_cost_yuan",
            value=cost_yuan,
            labels={"job_id": job_id}
        )

    def record_success(self, operation: str, success: bool, job_id: str):
        """记录成功/失败指标"""
        self._insert_metric(
            name=f"{operation}_success",
            value=1.0 if success else 0.0,
            labels={"job_id": job_id}
        )

    def _insert_metric(self, name: str, value: float, labels: dict):
        """插入指标数据"""
        import json
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("""
        INSERT INTO metrics (name, value, labels_json)
        VALUES (?, ?, ?)
        """, (name, value, json.dumps(labels)))
        conn.commit()
        conn.close()

    def get_avg_latency(self, operation: str, hours: int = 24) -> float:
        """获取平均延迟"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("""
        SELECT AVG(value) FROM metrics
        WHERE name = ? AND created_at > datetime('now', ?)
        """, (f"{operation}_latency_ms", f"-{hours} hours"))
        result = cursor.fetchone()[0]
        conn.close()
        return result or 0.0

    def get_total_cost(self, hours: int = 24) -> float:
        """获取总成本"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("""
        SELECT SUM(value) FROM metrics
        WHERE name LIKE '%_cost_yuan' AND created_at > datetime('now', ?)
        """, (f"-{hours} hours",))
        result = cursor.fetchone()[0]
        conn.close()
        return result or 0.0

    def get_failure_rate(self, operation: str, hours: int = 24) -> float:
        """获取失败率"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        cursor.execute("""
        SELECT COUNT(*) FROM metrics
        WHERE name = ? AND created_at > datetime('now', ?)
        """, (f"{operation}_success", f"-{hours} hours"))
        total = cursor.fetchone()[0]

        cursor.execute("""
        SELECT COUNT(*) FROM metrics
        WHERE name = ? AND value = 0 AND created_at > datetime('now', ?)
        """, (f"{operation}_success", f"-{hours} hours"))
        failures = cursor.fetchone()[0]

        conn.close()
        return failures / total if total > 0 else 0.0
```

#### 2.2.2 告警管理器

```python
# src/jobpilot/monitoring/alert_manager.py

from dataclasses import dataclass
from typing import Callable
from datetime import datetime

@dataclass
class AlertRule:
    """告警规则"""
    name: str
    condition: Callable[[], bool]
    message: str
    severity: str  # info, warning, critical

class AlertManager:
    """告警管理器"""

    def __init__(self, metrics_collector):
        self.metrics = metrics_collector
        self.rules: list[AlertRule] = []
        self.alert_history: list[dict] = []

    def add_rule(self, rule: AlertRule):
        """添加告警规则"""
        self.rules.append(rule)

    def check_alerts(self) -> list[dict]:
        """检查所有告警规则"""
        triggered = []

        for rule in self.rules:
            if rule.condition():
                alert = {
                    "name": rule.name,
                    "message": rule.message,
                    "severity": rule.severity,
                    "timestamp": datetime.now()
                }
                triggered.append(alert)
                self.alert_history.append(alert)

        return triggered

    def setup_default_rules(self):
        """设置默认告警规则"""
        # Claim 提取延迟告警
        self.add_rule(AlertRule(
            name="claim_extraction_latency",
            condition=lambda: self.metrics.get_avg_latency("claim_extraction", 1) > 10000,
            message="Claim 提取平均延迟超过 10 秒",
            severity="warning"
        ))

        # 月度成本告警
        self.add_rule(AlertRule(
            name="monthly_cost",
            condition=lambda: self.metrics.get_total_cost(720) > 40,  # 30 天
            message="月度总成本超过 ¥40",
            severity="critical"
        ))

        # LLM 调用失败率告警
        self.add_rule(AlertRule(
            name="llm_failure_rate",
            condition=lambda: self.metrics.get_failure_rate("claim_extraction", 24) > 0.05,
            message="Claim 提取 LLM 调用失败率超过 5%",
            severity="critical"
        ))
```

```python
# src/jobpilot/app/pages/monitoring.py

import streamlit as st
from jobpilot.monitoring.metrics_collector import MetricsCollector
from jobpilot.monitoring.alert_manager import AlertManager

def show_monitoring_page():
    """监控面板页面"""
    st.title("📊 扩展技能监控")

    metrics = MetricsCollector("data/offer_radar.db")
    alerts = AlertManager(metrics)
    alerts.setup_default_rules()

    # 概览指标
    col1, col2, col3, col4 = st.columns(4)

    with col1:
        avg_latency = metrics.get_avg_latency("claim_extraction", 24)
        st.metric("Claim 提取延迟", f"{avg_latency:.0f}ms")

    with col2:
        total_cost = metrics.get_total_cost(24)
        st.metric("今日成本", f"¥{total_cost:.2f}")

    with col3:
        failure_rate = metrics.get_failure_rate("claim_extraction", 24)
        st.metric("失败率", f"{failure_rate:.1%}")

    with col4:
        cache_stats = metrics.get_cache_stats()

    # 告警列表
    st.subheader("⚠️ 活跃告警")
    triggered = alerts.check_alerts()
    if triggered:
        for alert in triggered:
            st.warning(f"[{alert['severity']}] {alert['message']}")
    else:
        st.success("无活跃告警")

    # 成本趋势图
    st.subheader("💰 成本趋势")
    cost_data = metrics.get_cost_trend(days=7)
    st.line_chart(cost_data)

    # 延迟趋势图
    st.subheader("⏱️ 延迟趋势")
```

```python
# src/jobpilot/utils/logger.py

import logging
import json
from datetime import datetime

class StructuredLogger:
    """结构化日志器"""

    def __init__(self, name: str):
        self.logger = logging.getLogger(name)
        self.logger.setLevel(logging.INFO)

        handler = logging.FileHandler(f"logs/{name}.log")
        handler.setFormatter(logging.Formatter(
            '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        ))
        self.logger.addHandler(handler)

    def log_claim_extraction(self, job_id: str, claims_count: int,
                              gaps_count: int, latency_ms: float, cost: float):
        """记录 Claim 提取日志"""
        self.logger.info(json.dumps({
            "event": "claim_extraction",
            "job_id": job_id,
            "claims_count": claims_count,
            "gaps_count": gaps_count,
            "latency_ms": latency_ms,
            "cost": cost,
            "timestamp": datetime.now().isoformat()
        }))

    def log_gap_advisor(self, job_id: str, rewrites_count: int,
                         latency_ms: float, cost: float):
        """记录 Gap Advisor 日志"""
        self.logger.info(json.dumps({
            "event": "gap_advisor",
            "job_id": job_id,
            "rewrites_count": rewrites_count,
            "latency_ms": latency_ms,
            "cost": cost,
            "timestamp": datetime.now().isoformat()
        }))

    def log_interview_prep(self, job_id: str, questions_count: int,
                            latency_ms: float, cost: float):
        """记录 Interview Prep 日志"""
        self.logger.info(json.dumps({
            "event": "interview_prep",
            "job_id": job_id,
            "questions_count": questions_count,
            "latency_ms": latency_ms,
            "cost": cost,
            "timestamp": datetime.now().isoformat()
```

## 四、运维流程

### 4.1 版本发布流程

```
1. 开发阶段
   - 在 feature 分支开发
   - 运行单元测试 + 集成测试
   - 提交 PR

2. 代码审查
   - 至少 1 人审查
   - 检查测试覆盖率
   - 检查 Prompt 变更

3. 合并到 main
   - CI 自动运行测试
   - 自动部署到测试环境

4. 人工评测（里程碑节点）
   - 运行 50 条人工评测
   - 确认质量达标

5. 发布
   - 打 tag
   - 更新 CHANGELOG
   - 部署到生产环境
```

### 4.2 回滚流程

```
1. 发现问题
   - 监控告警
   - 用户反馈
   - 人工评测异常

2. 评估影响
   - 影响范围（哪些功能）
   - 严重程度（数据/成本/质量）

3. 决策回滚
   - 如果是 Prompt 问题：回滚 Prompt 文件
   - 如果是代码问题：回滚代码版本
   - 如果是数据问题：恢复数据库备份

4. 执行回滚
   - Git revert 或 checkout 到稳定版本
   - 重新部署

5. 验证
   - 运行测试
   - 检查监控指标

6. 复盘
   - 分析根因
   - 更新故障排查手册
```

### 4.3 紧急响应流程

```
1. 接收告警
   - 监控系统自动告警
   - 用户报告问题

2. 初步诊断
   - 查看日志
   - 检查监控面板
   - 确认影响范围

3. 临时缓解
   - 如果是 LLM 问题：切换到备用模型
   - 如果是成本问题：暂停自动任务
   - 如果是质量问题：禁用问题功能

4. 根因分析
   - 分析日志
   - 复现问题
   - 确定根因

5. 修复
   - 开发修复方案
   - 测试验证
   - 部署上线

6. 复盘
   - 更新故障排查手册
   - 优化监控告警
```

### 6.1 邮件告警配置

```python
# src/jobpilot/monitoring/email_alert.py

import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart

class EmailAlertSender:
    """邮件告警发送器"""

    def __init__(self, smtp_host: str, smtp_port: int,
                 username: str, password: str, recipients: list[str]):
        self.smtp_host = smtp_host
        self.smtp_port = smtp_port
        self.username = username
        self.password = password
        self.recipients = recipients

    def send_alert(self, subject: str, message: str, severity: str):
        """发送告警邮件"""
        msg = MIMEMultipart()
        msg['From'] = self.username
        msg['To'] = ', '.join(self.recipients)
        msg['Subject'] = f"[OfferRadar {severity.upper()}] {subject}"

        body = f"""
        <html>
        <body>
            <h2>OfferRadar 告警通知</h2>
            <p><strong>严重程度：</strong>{severity}</p>
            <p><strong>告警内容：</strong>{message}</p>
            <p><strong>处理建议：</strong>请登录监控面板查看详情</p>
        </body>
        </html>
        """

        msg.attach(MIMEText(body, 'html'))

        with smtplib.SMTP(self.smtp_host, self.smtp_port) as server:
            server.starttls()
            server.login(self.username, self.password)
            server.send_message(msg)
```

### 6.2 日志清理配置

```python
# src/jobpilot/scheduler/log_cleanup_job.py

from apscheduler.schedulers.background import BackgroundScheduler
from pathlib import Path
from datetime import datetime, timedelta

def schedule_log_cleanup(scheduler: BackgroundScheduler, log_dir: str):
    """调度日志清理任务"""
    def cleanup_logs():
        """清理 30 天前的日志文件"""
        log_path = Path(log_dir)
        cutoff = datetime.now() - timedelta(days=30)

        for log_file in log_path.glob("*.log"):
            if datetime.fromtimestamp(log_file.stat().st_mtime) < cutoff:
                log_file.unlink()

    # 每日凌晨 4 点清理日志
    scheduler.add_job(
        cleanup_logs,
        'cron',
        hour=4,
        id='log_cleanup',
        name='Log Cleanup'
    )
```

```python
# src/jobpilot/scheduler/backup_cleanup_job.py

from apscheduler.schedulers.background import BackgroundScheduler
from pathlib import Path
from datetime import datetime, timedelta

def schedule_backup_cleanup(scheduler: BackgroundScheduler, backup_dir: str):
    """调度备份清理任务"""
    def cleanup_backups():
        """清理 30 天前的备份文件"""
        backup_path = Path(backup_dir)
        cutoff = datetime.now() - timedelta(days=30)

        for backup_file in backup_path.glob("*.db"):
            if datetime.fromtimestamp(backup_file.stat().st_mtime) < cutoff:
                backup_file.unlink()

    # 每周日凌晨 5 点清理备份
    scheduler.add_job(
        cleanup_backups,
        'cron',
        day_of_week='sun',
        hour=5,
        id='backup_cleanup',
        name='Backup Cleanup'
    )
```
